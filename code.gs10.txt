function report(){

  if(l>1)rows=sh.getRange(2,1,l-1,H_MX.length).getValues();

  }).join('');

    }catch(err){return out({ok:false,error:err.message})}
  }
  if(!p.key)return HtmlService.createHtmlOutput(pintu());
  if(p.key!==SECRET_KEY)return HtmlService.createHtmlOutput(pintu('Kunci salah.'));
  AUTO_RELOAD=p.auto==='1';
